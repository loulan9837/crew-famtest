## 二、存储层：case_conversation_store.py

新建独立存储模块，**优先使用 SQLite**，仅在无 SQLite 环境下降级 JSON，与 `memory_store.py` 双端降级设计保持一致。

### 2.1 存储架构

| 后端 | 条件 | 说明 |
|------|------|------|
| SQLite | 默认 | 新建 `conversations` 与 `messages` 表，利用行级锁解决并发 |
| JSON | 降级 | 仅当 SQLite 不可用时使用 `output/case_conversations.json`，`write → rename` 原子写入 |

### 2.2 文件结构与 SQLite 表结构

```

├── _ensure_tables()   # 创建 conversations / messages 表
```

**SQLite 表结构示例**：

```sql
CREATE TABLE IF NOT EXISTS conversations (
    id TEXT PRIMARY KEY,
    title TEXT,
    run_record_id TEXT,
    prd_snapshot TEXT,
    cases_snapshot_md TEXT,
    created_at TEXT,
    updated_at TEXT,
    project_id TEXT
);

CREATE TABLE IF NOT EXISTS case_chat_messages (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    conv_id TEXT NOT NULL,
    role TEXT NOT NULL,
    content TEXT NOT NULL,
    timestamp TEXT,
    FOREIGN KEY (conv_id) REFERENCES conversations(id)
);
```

```python
def _render_conversation_history(T: dict):
    """渲染历史对话列表，含搜索、恢复、删除。"""
    with st.expander("对话历史", expanded=False):
        keyword = st.text_input(
            "搜索",
            key=_get_case_chat_key("history_search"),
            placeholder="按标题搜索历史对话...",
            label_visibility="collapsed",
        )
        try:
            from case_conversation_store import list_conversations, delete_conversation, get_conversation
        except ImportError:
            st.caption("对话存储模块未就绪")
            return

        convs = list_conversations(
            keyword=(keyword or "").strip(),
            limit=20,
            project_id=_get_current_project(),
        )
        if not convs:
            st.info("暂无历史对话")
            return

        delete_confirm = st.session_state.get(_get_case_chat_key("delete_confirm_id"))

        for conv in convs:
            cid = conv["id"]
            cols = st.columns([5, 1, 1])
            with cols[0]:
                title = conv.get("title", "无标题")
                msg_count = conv.get("message_count", 0)
                updated = conv.get("updated_at", "")
                st.markdown(f"**{title}**  \n{msg_count}条消息 · {updated}")

            with cols[1]:
                if st.button("恢复", key=f"conv_resume_{cid}"):
                    full = get_conversation(cid)
                    if full:
                        st.session_state[_get_case_chat_key("current_conv_id")] = cid
                        st.session_state[_get_case_chat_key("messages")] = full.get("messages", [])
                        st.session_state[_get_case_chat_key("prd_context")] = full.get("prd_snapshot", "")
                        st.session_state[_get_case_chat_key("cases_context")] = full.get("cases_snapshot_md", "")
                        st.rerun()

            with cols[2]:
                # 手动删除（对话被真正移除的唯一主动途径，需二次确认）
                if delete_confirm == cid:
                    if st.button("确认删除", key=f"conv_del_confirm_{cid}", type="primary"):
                        delete_conversation(cid)
                        st.session_state[_get_case_chat_key("delete_confirm_id")] = None
                        # 若删除的恰好是当前活跃对话，清空工作区
                        if st.session_state.get(_get_case_chat_key("current_conv_id")) == cid:
                            st.session_state[_get_case_chat_key("current_conv_id")] = None
                            st.session_state[_get_case_chat_key("messages")] = []
                            st.session_state[_get_case_chat_key("prd_context")] = ""
                            st.session_state[_get_case_chat_key("cases_context")] = ""
                        st.rerun()
                else:
                    if st.button("删除", key=f"conv_del_{cid}"):
                        st.session_state[_get_case_chat_key("delete_confirm_id")] = cid
                        st.rerun()
```

### 3.5 动态变量注入：解析下一用例ID

```python
import re

def _parse_next_case_id_from_md(md: str) -> str | None:
    """从用例 Markdown 快照中解析当前最大 ID，返回下一 ID（如 FAM-016）。
    支持 FAM-001、ACT-003、TC-012 等格式。若无匹配则返回 None。"""
    if not (md or "").strip():
        return None
    # 匹配 | FAM-001 | 或 | ACT-003 | 等表格首列
    matches = re.findall(r"^\|\s*([A-Za-z]+-\d+)\s*\|", md, re.MULTILINE)
    if not matches:
        return None
    def _key(m):
        prefix, num = m.rsplit("-", 1)
        return (prefix, int(num) if num.isdigit() else 0)
    last = max(matches, key=_key)
    p, n = last.rsplit("-", 1)
    next_n = int(n) + 1 if n.isdigit() else 1
    return f"{p}-{next_n:03d}"
```

```python
def _send_message(user_msg: str, defaults: dict):
    """发送用户消息、调用Agent、存储双方消息。"""
    from datetime import datetime

    prd_ctx = st.session_state.get(_get_case_chat_key("prd_context"), "")
    cases_ctx = st.session_state.get(_get_case_chat_key("cases_context"), "")
    messages = st.session_state.get(_get_case_chat_key("messages"), [])
    conv_id = st.session_state.get(_get_case_chat_key("current_conv_id"))

    now_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    user_entry = {"role": "user", "content": user_msg, "timestamp": now_str}
    messages.append(user_entry)
    st.session_state[_get_case_chat_key("messages")] = messages

    # 首条消息时自动创建对话记录
    if not conv_id:
        from case_conversation_store import create_conversation
        title = (user_msg[:30] + "...") if len(user_msg) > 30 else user_msg
        conv_id = create_conversation(
            title=title,
            prd_snapshot=prd_ctx,
            cases_snapshot_md=cases_ctx,
            project_id=_get_current_project(),
        )
        st.session_state[_get_case_chat_key("current_conv_id")] = conv_id

    # 持久化用户消息
    from case_conversation_store import append_message
    if conv_id:
        append_message(conv_id, "user", user_msg)

    # 调用Agent
    try:
        reply = _call_case_chat_agent(
            user_msg=user_msg,
            prd_context=prd_ctx,
            cases_context=cases_ctx,
            history=messages[:-1],  # 不含本次用户消息（已在prompt中）
            defaults=defaults,
        )
    except Exception as e:
        reply = f"Agent调用失败：{e}"

    now_str2 = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    assistant_entry = {"role": "assistant", "content": reply, "timestamp": now_str2}
    messages.append(assistant_entry)
    st.session_state[_get_case_chat_key("messages")] = messages

    # 持久化Agent回复
    if conv_id:
        append_message(conv_id, "assistant", reply)
```

### 4.2 Prompt构造逻辑

```python
def _build_case_chat_prompt(
    user_message: str,
    prd_context: str,
    cases_context: str,
    conversation_history: list[dict] | None = None,
    project_context: str = "",
) -> str:
    parts: list[str] = []

    # System角色
    parts.append("你是一位资深的测试用例顾问（Test Case Advisor），拥有丰富的测试开发经验。")
    parts.append("")

    # 上下文：完整传入，不截断（Gemini 1.5 Pro 等具备超长上下文；可选接入 Context Caching）
    prd_full = (prd_context or "").strip()
    cases_full = (cases_context or "").strip()
    if prd_full:
        parts.append("=== 需求文档 ===")
        parts.append(prd_full)
        parts.append("")

    if cases_full:
        parts.append("=== 现有用例表 ===")
        parts.append(cases_full)
        parts.append("")

    if project_context:
        proj = (project_context or "")[:5000]  # 项目记忆可适度截断
        parts.append("=== 项目记忆 ===")
        parts.append(proj)
        parts.append("")

    # 产出规范（含 <case_table> 物理隔离，为未来一键合入留锚点）
    parts.append("=== 产出规范（输出用例时必须遵守）===")
    parts.append("1. 表头固定：ID | 模块 | 测试项 | 优先级 | 前置条件 | 操作步骤 | 预期结果")
    parts.append("2. 中英/数字边界不得出现空格（如：点击Save按钮）")
    parts.append("3. 预期结果绝对去动词化，不得出现"点击""查看"等动作词")
    parts.append("4. 新增用例ID从当前最大ID+1开始，修改用例保留原始ID")
    parts.append("5. 【关键】当回复中包含具体测试用例表格时，必须且只能将表格放在 <case_table> 与 </case_table> 标签内部；解释性文字写在标签外")
    parts.append("6. 回复时先简要说明分析思路（2-3句），再给出具体用例表格或修改建议")
    parts.append("7. 发现修改请求与现有逻辑冲突时，必须主动提醒")
    parts.append("")

    # 对话历史（滑动窗口：最近20条）
    history = (conversation_history or [])[-20:]
    if history:
        parts.append("=== 对话历史 ===")
        for msg in history:
            role_label = "用户" if msg.get("role") == "user" else "顾问"
            parts.append(f"[{role_label}] {msg.get('content', '')}")
        parts.append("")

    # 当前问题
    parts.append("=== 用户当前问题 ===")
    parts.append(user_message)

    return "\n".join(parts)
```

> **Context Caching（可选增强）**：若接入 Gemini Context Caching，可将 `prd_full` 与 `cases_full` 作为 Cached Content 挂载，每次对话仅消耗新增 Token，降低成本与延迟。

**表格提取（未来一键合入）**：底层可通过正则精准截取 `content` 中的 `<case_table>...</case_table>` 内容，避免与自然语言混排导致解析崩溃：

```python
import re
def _extract_case_table_from_reply(content: str) -> str | None:
    """从 Agent 回复中提取 <case_table> 内的 Markdown 表格，供未来一键合入使用。"""
    m = re.search(r"<case_table>\s*([\s\S]*?)\s*</case_table>", content, re.IGNORECASE)
    return m.group(1).strip() if m else None
```

```python
def chat_with_cases_agent(
    user_message: str,
    prd_context: str,
    cases_context: str,

    project_context: str = "",

    )
```

```python
def _render_module_case_chat(T: dict, defaults: dict):
    """工作台模块：用例对话。"""
    st.subheader(_get_text(T, "case_chat_tab.section_title") or "用例对话")
    st.caption(
        _get_text(T, "case_chat_tab.section_desc")
        or "与Agent多轮对话，逐步补充、修改和优化测试用例。"
    )

    # --- 对话历史区 ---
    _render_conversation_history(T)

    st.divider()

    conv_id = st.session_state.get(_get_case_chat_key("current_conv_id"))

    # --- 无活跃对话：展示来源选择 ---
    if not conv_id:
        prd_ctx, cases_ctx, run_record_id = _render_new_conversation_setup(T, defaults)
        st.session_state[_get_case_chat_key("prd_context")] = prd_ctx
        st.session_state[_get_case_chat_key("cases_context")] = cases_ctx
        if not prd_ctx and not cases_ctx:
            st.info("请选择需求来源后开始对话。")
            return

    # --- 有活跃对话/已选择来源：展示对话区 ---
    if conv_id:
        # 快照时间提示
        try:
            from case_conversation_store import get_conversation
            conv_data = get_conversation(conv_id)
            if conv_data:
                created = conv_data.get("created_at", "")
                st.caption(f"对话创建于 {created}")
        except ImportError:
            pass

        # 结束当前对话按钮（保存并归档，不删除）
        if st.button(
            _get_text(T, "case_chat_tab.end_conversation_btn") or "结束当前对话",
            key="case_chat_end_conv",
        ):
            # 结束 = 保存归档，对话记录保留在历史中可随时恢复
            # 确保最后一轮内容已持久化（append_message在发送时已调用，此处做兜底）
            try:
                from case_conversation_store import get_conversation
                if conv_id and get_conversation(conv_id):
                    pass  # 已持久化，无需额外操作
            except ImportError:
                pass
            # 仅清空当前工作区session_state，不从JSON中删除任何记录
            st.session_state[_get_case_chat_key("current_conv_id")] = None
            st.session_state[_get_case_chat_key("messages")] = []
            st.session_state[_get_case_chat_key("prd_context")] = ""
            st.session_state[_get_case_chat_key("cases_context")] = ""
            st.rerun()

    _render_chat_area(T, defaults)
```
